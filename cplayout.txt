'use client';

import React, { useState } from 'react';
import Link from 'next/link';
import { usePathname, useRouter, useSearchParams } from 'next/navigation';
import { 
  FolderKanban, 
  FileSignature, 
  FileText, 
  Users, 
  Bell, 
  LogOut, 
  Menu, 
  X,
  LayoutDashboard
} from 'lucide-react';
import styles from './client-portal-layout.module.css';

export default function ClientPortalLayout({ children }: { children: React.ReactNode }) {
  const pathname = usePathname();
  const router = useRouter();
  const searchParams = useSearchParams();
  const [isSidebarOpen, setIsSidebarOpen] = useState(false);

  // Auth pages in client portal render without sidebar/topbar
  const isAuthPage = pathname === '/client-portal' || pathname === '/client-portal/login' || pathname === '/client-portal/accept-invite';
  if (isAuthPage) {
    return <>{children}</>;
  }

  const clientName = searchParams.get('client') || 'Atlas Corp';

  const navItems = [
    { name: 'Projects & Hub', href: `/client-portal/dashboard?client=${encodeURIComponent(clientName)}`, icon: LayoutDashboard },
    { name: 'All Deliverables', href: '/client-portal/projects', icon: FolderKanban },
    { name: 'Contracts & NDAs', href: '/client-portal/contracts', icon: FileSignature },
    { name: 'Invoices & Billing', href: '/client-portal/invoices', icon: FileText },
    { name: 'Company Profile', href: '/client-portal/profile', icon: Users },
  ];

  const handleLogout = () => {
    router.push('/client-portal/login');
  };

  return (
    <div className={styles.portalContainer}>
      {/* â”€â”€â”€ Client Sidebar (On the left) â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€ */}
      <aside className={`${styles.sidebar} ${isSidebarOpen ? styles.sidebarOpen : ''}`}>
        <div className={styles.sidebarHeader}>
          <div className={styles.brandRow}>
            <div className={styles.logoBadge}>AP</div>
            <div className={styles.brandText}>
              <span className={styles.logoTitle}>AETHER OS</span>
              <span className={styles.portalSubTag}>CLIENT PORTAL</span>
            </div>
          </div>
          <button 
            className={styles.closeBtnMobile} 
            onClick={() => setIsSidebarOpen(false)}
          >
            <X size={20} />
          </button>
        </div>

        <nav className={styles.nav}>
          {navItems.map((item) => {
            const isActive = pathname === item.href.split('?')[0];
            const Icon = item.icon;
            return (
              <Link 
                key={item.name} 
                href={item.href}
                className={`${styles.navItem} ${isActive ? styles.navActive : ''}`}
                onClick={() => setIsSidebarOpen(false)}
              >
                <Icon size={18} />
                <span>{item.name}</span>
              </Link>
            );
          })}
        </nav>

        <div className={styles.sidebarFooter}>
          <div className={styles.userCard}>
            <div className={styles.userAvatar}>
              {clientName.slice(0, 2).toUpperCase()}
            </div>
            <div className={styles.userInfo}>
              <span className={styles.userName}>{clientName}</span>
              <span className={styles.userRole}>Verified Client</span>
            </div>
          </div>
          <button className={styles.logoutBtn} onClick={handleLogout}>
            <LogOut size={16} /> Sign Out
          </button>
        </div>
      </aside>

      {/* Overlay for mobile drawer */}
      {isSidebarOpen && (
        <div 
          className={styles.backdrop} 
          onClick={() => setIsSidebarOpen(false)} 
        />
      )}

      {/* â”€â”€â”€ Main Content Area with Top Navbar â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€ */}
      <div className={styles.mainWrapper}>
        {/* Top Navbar */}
        <header className={styles.topbar}>
          <div className={styles.topbarLeft}>
            <button 
              className={styles.menuBtn} 
              onClick={() => setIsSidebarOpen(true)}
              aria-label="Open Navigation"
            >
              <Menu size={22} />
            </button>
            <div className={styles.topbarBrand}>
              <div className={styles.topbarLogoCircle}>AP</div>
              <span className={styles.topbarBrandText}>AETHER OS</span>
            </div>
          </div>

          <div className={styles.topbarRight}>
            <button className={styles.bellBtn} title="Notifications">
              <Bell size={19} />
              <span className={styles.notifDot} />
            </button>
          </div>
        </header>

        {/* Page Content */}
        <main className={styles.content}>
          {children}
        </main>
      </div>
    </div>
  );
}
